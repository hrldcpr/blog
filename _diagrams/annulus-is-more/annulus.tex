\documentclass{article}
\usepackage{tikz}
\usepackage[margin=1in]{geometry}

\begin{document}
\thispagestyle{empty}
\begin{tikzpicture}
    % Define parameters
    \def\r{1}      % inner radius
    \def\d{0.5}    % width

    % Draw the annulus
    \fill[lightgray] (0, 0) circle (\r + \d);
    \fill[white] (0, 0) circle (\r);

    % Draw the boundaries
    \draw[thick] (0, 0) circle (\r);
    \draw[thick] (0, 0) circle (\r + \d);

    % Add labels
    \draw[-] (0, 0) -- ({\r/sqrt(2)}, {\r/sqrt(2)});
    \node[above] at ({\r/sqrt(2)/3}, {\r/sqrt(2)/3}) {$r$};

    \draw[-] (\r, 0) -- (\r + \d, 0);
    \node[above] at (\r + \d/2, 0) {$d$};

    % Add center point
    \fill (0,0) circle (0.05);
\end{tikzpicture}
\end{document}
