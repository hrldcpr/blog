\documentclass{article}
\usepackage{tikz}
\usepackage[margin=1in]{geometry}

\begin{document}
\thispagestyle{empty}
\begin{tikzpicture}
  \def\r{1}
  \def\d{0.5}
  \pgfmathsetmacro{\rminor}{0.3*\r}  % minor axis for 3D effect

  \begin{scope}
    \draw[fill=lightgray] (0, 0) circle (\r + \d);
    \draw[fill=white] (0, 0) circle (\r);
    \fill (0,0) circle (0.05);

    \draw[-] (0, 0) -- node [above] {$r$} ({\r/sqrt(2)}, {\r/sqrt(2)});
    \draw[-] (\r, 0) -- node [above] {$d$} (\r + \d, 0);

    \node[right] at (\r+\d+0.1, 0) {$=$}; % TODO auto-position to right of outer circle
  \end{scope}

  \begin{scope}[shift={(\r+\d+\r+0.7, 0)}] % cylinder
    \draw[-] (0,0) ellipse ({\r} and {\rminor});
    \draw[-] (0,\d) ellipse ({\r} and {\rminor});
    \draw[-] (-\r,0) -- (-\r,\d);
    \draw[-] (\r,0) -- (\r,\d);
    \fill[lightgray] (-\r,0) -- (-\r,\d) arc (180:360:{\r} and {\rminor}) -- (\r,0) arc (180:360:{-\r} and {\rminor});
    \fill (0,\d) circle (0.05);

    \draw[-] (0,\d) -- node [above=-2] {$r$} (\r,\d);
    \draw[<->] (\r+0.1,0) -- node [right] {$d$} (\r+0.1,\d);

    \node[right] at (\r+0.3, 0) {$+$};
  \end{scope}

  \begin{scope}[shift={(\r+\d+\r+\r+0.5+\d+0.9, \d/2)}] % circle
    \draw[fill=lightgray] (0, 0) circle (\d);
    \draw[-] (0, 0) -- node [above=-1] {$d$} (\d, 0);
    \fill (0,0) circle (0.05);
  \end{scope}

\end{tikzpicture}
\end{document}
